module tb;
    // Signals
    logic        pclk;
    logic        prstn;
    logic [4:0]  paddr;
    logic        psel;
    logic        penable;
    logic        pwrite;
    logic [31:0] pwdata;
    logic        pready;
    logic        trans;
    logic [31:0] prdata;

    // Instantiate DUT
    dut uut (.*);

    // Clock Generation
    initial pclk = 0;
    always #5 pclk = ~pclk;

    // Simulation Logic
    initial begin
        // --- 1. Reset ---
        prstn = 0; psel = 0; penable = 0; pwrite = 0; 
        paddr = 0; pwdata = 0; pready = 0; trans = 0;
        #20 prstn = 1;
        repeat(2) @(posedge pclk);

        // --- 2. WAIT-STATE WRITE ---
        // Setup Phase
        @(posedge pclk);
        psel <= 1; paddr <= 5'd5; pwdata <= 32'hAAAA_BBBB; pwrite <= 1; trans <= 1;
        // Access Phase starts
        @(posedge pclk);
        penable <= 1; pready <= 0; // Slave is "busy"
        repeat(2) @(posedge pclk); // Stall for 2 cycles
        pready <= 1;               // Slave is "ready"
        @(posedge pclk);
        psel <= 0; penable <= 0; pready <= 0; trans <= 0;

        

        repeat(2) @(posedge pclk);

        // --- 3. CONTINUOUS (BACK-TO-BACK) WRITES ---
        // Transfer 1: Setup
        @(posedge pclk);
        psel <= 1; paddr <= 5'd1; pwdata <= 32'h1111; pwrite <= 1; trans <= 1;
        // Transfer 1: Access / Transfer 2: Setup
        @(posedge pclk);
        penable <= 1; pready <= 1; 
        paddr <= 5'd2; pwdata <= 32'h2222; // Change addr/data while in Access
        // Transfer 2: Access
        @(posedge pclk);
        penable <= 1; pready <= 1; 
        @(posedge pclk);
        psel <= 0; penable <= 0; pready <= 0; trans <= 0;

        

        // --- 4. TRANSACTION BREAK (IDLE) ---
        repeat(4) @(posedge pclk); 

        // --- 5. WAIT-STATE READ ---
        // Setup
        @(posedge pclk);
        psel <= 1; paddr <= 5'd5; pwrite <= 0; trans <= 1;
        // Access
        @(posedge pclk);
        penable <= 1; pready <= 0; // Wait State
        repeat(1) @(posedge pclk);
        pready <= 1; 
        @(posedge pclk);
        $display("Captured Read Data: %h", prdata);
        psel <= 0; penable <= 0; pready <= 0; trans <= 0;

        

        #50 $finish;
    end

    // Monitor for debugging
    initial begin
        $monitor("Time:%0t | State:%s | PADDR:%0d | PSEL:%b | PENABLE:%b | PREADY:%b | DATA:%h", 
                 $time, uut.state.name(), paddr, psel, penable, pready, (pwrite ? pwdata : prdata));
    end

endmodule
